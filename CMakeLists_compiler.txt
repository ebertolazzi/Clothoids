# Shared PINS compiler policy. Keep this file byte-for-byte identical in each
# PINS component and in PINS2/sources/src.
# Include before project() so CMP0091 and the MSVC runtime are initialized in
# time; call pins_configure_compiler() immediately after project().
if(POLICY CMP0091)
  cmake_policy(SET CMP0091 NEW)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>DLL")
endif()
if(POLICY CMP0141)
  cmake_policy(SET CMP0141 NEW)
  set(CMAKE_MSVC_DEBUG_INFORMATION_FORMAT "$<$<CONFIG:Debug,RelWithDebInfo>:Embedded>")
endif()

set(PINS_EIGEN_VERSION "5.0.1" CACHE STRING "Pinned Eigen version shared by PINS" FORCE)
set(CYCLIC_REDUCTION_EIGEN_VERSION "${PINS_EIGEN_VERSION}" CACHE STRING "Eigen version" FORCE)
set(SPLINES_EIGEN_VERSION "${PINS_EIGEN_VERSION}" CACHE STRING "Eigen version" FORCE)
set(UTILS_EIGEN_VERSION "${PINS_EIGEN_VERSION}" CACHE STRING "Eigen version" FORCE)

function(pins_configure_compiler)
  if(NOT CMAKE_CXX_COMPILER_ID MATCHES "^(GNU|Clang|AppleClang|MSVC|IntelLLVM)$")
    message(FATAL_ERROR "Unsupported C++ compiler: ${CMAKE_CXX_COMPILER_ID}")
  endif()

  # These cache overrides intentionally defeat stale opt-in values from older
  # builds. Eigen's public fixed-size types must have the same layout in every
  # PINS library and in code generated by XOptima.
  foreach(_pins_native_option IN ITEMS
      PINS_USE_NATIVE_ARCH
      ASTRO_USE_NATIVE_ARCH
      CLOTHOIDS_USE_NATIVE_ARCH
      CYCLIC_REDUCTION_USE_NATIVE_ARCH
      GENERIC_CONTAINER_USE_NATIVE_ARCH
      REGULARIZED_USE_NATIVE_ARCH
      SPLINES_USE_NATIVE_ARCH
      UTILS_USE_NATIVE_ARCH)
    set(${_pins_native_option} OFF CACHE BOOL "Disabled by the shared PINS ABI policy" FORCE)
  endforeach()

  set(CMAKE_CXX_STANDARD 20 PARENT_SCOPE)
  set(CMAKE_CXX_STANDARD_REQUIRED ON PARENT_SCOPE)
  set(CMAKE_CXX_EXTENSIONS OFF PARENT_SCOPE)

  if(CMAKE_SIZEOF_VOID_P EQUAL 8)
    set(_pins_target_bits 64)
  elseif(CMAKE_SIZEOF_VOID_P EQUAL 4)
    set(_pins_target_bits 32)
  else()
    set(_pins_target_bits unknown)
  endif()
  if(CMAKE_SYSTEM_PROCESSOR)
    set(_pins_target_processor "${CMAKE_SYSTEM_PROCESSOR}")
  elseif(CMAKE_VS_PLATFORM_NAME)
    set(_pins_target_processor "${CMAKE_VS_PLATFORM_NAME}")
  else()
    set(_pins_target_processor unknown)
  endif()
  string(TOLOWER "${_pins_target_processor}" _pins_target_processor_lower)

  # Strip CPU-specific and explicit MSVC-runtime options from global flags.
  # Keep width selectors such as -m64: they choose the ABI, not a SIMD level.
  # This part deliberately runs on every call because PINS invokes the function
  # again after importing its legacy generated configuration.
  foreach(_pins_flag_var IN ITEMS
      CMAKE_C_FLAGS CMAKE_CXX_FLAGS
      CMAKE_C_FLAGS_DEBUG CMAKE_C_FLAGS_RELEASE CMAKE_C_FLAGS_RELWITHDEBINFO CMAKE_C_FLAGS_MINSIZEREL
      CMAKE_CXX_FLAGS_DEBUG CMAKE_CXX_FLAGS_RELEASE CMAKE_CXX_FLAGS_RELWITHDEBINFO CMAKE_CXX_FLAGS_MINSIZEREL)
    if(DEFINED ${_pins_flag_var})
      set(_pins_flags "${${_pins_flag_var}}")
      string(REGEX REPLACE "(^|[ \t])[-/]arch:[^ \t]+" " " _pins_flags "${_pins_flags}")
      string(REGEX REPLACE "(^|[ \t])-m(arch|tune|cpu)=[^ \t]+" " " _pins_flags "${_pins_flags}")
      string(REGEX REPLACE "(^|[ \t])-m(avx[^ \t]*|sse[^ \t]*)" " " _pins_flags "${_pins_flags}")
      string(REGEX REPLACE "(^|[ \t])/(MDd?|MTd?)([ \t]|$)" " " _pins_flags "${_pins_flags}")
      string(REGEX REPLACE "[ \t]+" " " _pins_flags "${_pins_flags}")
      string(STRIP "${_pins_flags}" _pins_flags)
      set(${_pins_flag_var} "${_pins_flags}" CACHE STRING "Sanitized by shared PINS compiler policy" FORCE)
      set(${_pins_flag_var} "${_pins_flags}" PARENT_SCOPE)
    endif()
  endforeach()

  # Directory properties avoid duplicating options when the sanitizing pass is
  # intentionally repeated. Project-specific warning suppressions may still be
  # added by each component, but the language/ABI/optimization baseline is one.
  get_property(_pins_policy_added DIRECTORY PROPERTY PINS_COMPILER_POLICY_OPTIONS_ADDED)
  if(NOT _pins_policy_added)
    set_property(DIRECTORY PROPERTY PINS_COMPILER_POLICY_OPTIONS_ADDED TRUE)

    add_compile_definitions(
      EIGEN_MAX_ALIGN_BYTES=16
      EIGEN_MAX_STATIC_ALIGN_BYTES=16
      "$<$<CONFIG:Release,RelWithDebInfo,MinSizeRel>:NDEBUG>"
      "$<$<CONFIG:Release,RelWithDebInfo,MinSizeRel>:EIGEN_NO_DEBUG>"
    )

    if(MSVC)
      add_compile_options(
        "$<$<COMPILE_LANGUAGE:C,CXX>:/W4>"
        "$<$<COMPILE_LANGUAGE:C,CXX>:/utf-8>"
        "$<$<COMPILE_LANGUAGE:CXX>:/permissive->"
        "$<$<COMPILE_LANGUAGE:CXX>:/EHsc>"
        "$<$<COMPILE_LANGUAGE:CXX>:/bigobj>"
        "$<$<COMPILE_LANGUAGE:CXX>:/Zc:__cplusplus>"
        "$<$<AND:$<CONFIG:Release,RelWithDebInfo,MinSizeRel>,$<COMPILE_LANGUAGE:C,CXX>>:/O2>"
        "$<$<AND:$<CONFIG:Debug>,$<COMPILE_LANGUAGE:C,CXX>>:/Od>"
        "$<$<AND:$<CONFIG:Debug>,$<COMPILE_LANGUAGE:C,CXX>>:/Ob0>"
      )
    else()
      add_compile_options(
        "$<$<COMPILE_LANGUAGE:C,CXX>:-Wall>"
        "$<$<COMPILE_LANGUAGE:C,CXX>:-Wextra>"
        "$<$<AND:$<CONFIG:Release,RelWithDebInfo,MinSizeRel>,$<COMPILE_LANGUAGE:C,CXX>>:-O2>"
        "$<$<AND:$<CONFIG:Debug>,$<COMPILE_LANGUAGE:C,CXX>>:-O0>"
      )

      # MinGW/LLVM on 64-bit Windows gets an explicit portable x86-64 target.
      # Native MSVC x64 is already generic when no /arch option is present.
      if(WIN32 AND _pins_target_bits EQUAL 64
         AND _pins_target_processor_lower MATCHES "^(x86_64|amd64|x64)$")
        add_compile_options(
          "$<$<COMPILE_LANGUAGE:C,CXX>:-march=x86-64>"
          "$<$<COMPILE_LANGUAGE:C,CXX>:-mtune=generic>"
        )
      endif()
    endif()
  endif()

  message(STATUS
    "PINS compiler policy: ${CMAKE_CXX_COMPILER_ID} ${CMAKE_CXX_COMPILER_VERSION}; "
    "target=${_pins_target_processor}/${_pins_target_bits}; Eigen=${PINS_EIGEN_VERSION}; "
    "alignment=16; native CPU flags=OFF")
endfunction()
